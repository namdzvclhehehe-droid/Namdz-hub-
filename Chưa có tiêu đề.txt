local function SaveConfig()
	pcall(function()
		if not isfolder(SAVE_FOLDER) then

local function LoadConfig()
	pcall(function()
		if isfile(SAVE_FILE) then
			local decoded = HttpService:JSONDecode(readfile(SAVE_FILE))
			for k, v in pairs(decoded) do
				if Config[k] ~= nil then
					Config[k] = v
				end
			end
			if not decoded.HideNameDefaultV2 then
				Config.HideNameAndID = true
				Config.HideNameDefaultV2 = true
			end
			if type(decoded.WebhookUrl) == "string" and decoded.WebhookUrl ~= "" then
				Config.WebhookUrl = decoded.WebhookUrl
			elseif type(decoded.BossWebhookUrl) == "string" and decoded.BossWebhookUrl ~= "" then
				Config.WebhookUrl = decoded.BossWebhookUrl
			elseif type(decoded.WeatherWebhookUrl) == "string" and decoded.WeatherWebhookUrl ~= "" then
				Config.WebhookUrl = decoded.WeatherWebhookUrl
			end
			Config.BossWebhookUrl = Config.WebhookUrl
			Config.WeatherWebhookUrl = Config.WebhookUrl
		end
	end)
end

local function GetIslandFarmCFrame(islandKey)
	local key = islandKey or Config.SelectedFarmIsland or ""
	if IslandFarmPositions[key] then
		return IslandFarmPositions[key]
	end
	local cleanKey = tostring(key):gsub("%s*|%s*%d+", ""):gsub("^%s+", ""):gsub("%s+$", "")
	if IslandFarmPositions[cleanKey] then
		return IslandFarmPositions[cleanKey]
	end
	local lowerClean = string.lower(cleanKey)
	for name, cf in pairs(IslandFarmPositions) do
		local lName = string.lower(name)
		if
			lowerClean == lName
			or string.find(lowerClean, lName, 1, true)
			or string.find(lName, lowerClean, 1, true)
		then
			return cf
		end
	end
	return nil
end

	elseif string.find(lower, "sun") or string.find(lower, "clear") then

	end

end

local function SendDiscordWebhook(rawUrl, title, description, customColor, extraFields)
	local url = tostring(rawUrl or ""):gsub("%s+", "")
	if url == "" then
		return false
	end
	if not (string.find(url, "discord.com/api/webhooks") or string.find(url, "discordapp.com/api/webhooks") or string.find(url, "webhook")) then
		return false
	end
	local req = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
	if not req then
		return false
	end
	local fields = {}
	if type(extraFields) == "table" then
		for _, f in ipairs(extraFields) do
			table.insert(fields, f)
		end
	end
	table.insert(fields, {
		name = "Server JobId",
		value = tostring(game.JobId),
		inline = false,
	})
	local payload = {
		embeds = {
			{
				title = title,
				description = description,
				color = customColor or 65280,
				fields = fields,
				footer = {
					text = "namdz Hub",
				},
				timestamp = DateTime.now():ToIsoDate(),
			},
		},
	}
	local ok, res = pcall(function()
		return req({
			Url = url,
			Method = "POST",
			Headers = {
				["Content-Type"] = "application/json",
			},
			Body = HttpService:JSONEncode(payload),
		})
	end)
	if ok and res then
		local sc = res.StatusCode or res.status_code or res.Status
		if sc and tonumber(sc) and tonumber(sc) >= 200 and tonumber(sc) < 300 then
			return true
		end
		if not sc then
			return true
		end
	end
	return false
end

local function GetNearestIslandName(pos)
	if not pos then
		return "Unknown"
	end
	local bestName = "Unknown"
	local bestDist = math.huge
	for name, cf in pairs(IslandFarmPositions) do
		local d = (cf.Position - pos).Magnitude
		if d < bestDist then
			bestDist = d
			bestName = name
		end
	end
	return bestName
end

local function GetChuPrompt()
	local npcFolder = workspace:FindFirstChild("NPC")
	local funcFolder = npcFolder and npcFolder:FindFirstChild("Function")
	local chu = funcFolder and funcFolder:FindFirstChild("Chu Tianjiao")
	if chu then
		local p = chu:FindFirstChildWhichIsA("ProximityPrompt", true)
		if p then
			return p
		end
	end
	for _, desc in ipairs(workspace:GetDescendants()) do
		if desc:IsA("ProximityPrompt") then
			local pName = string.lower(tostring(desc.ObjectText or (desc.Parent and desc.Parent.Name) or ""))
			local aName = string.lower(tostring(desc.ActionText or ""))
			if string.find(pName, "chu", 1, true) or string.find(aName, "talk", 1, true) then
				return desc
			end
		end
	end
	local direct = workspace:FindFirstChild("Chu Tianjiao", true)
	if direct then
		local p = direct:FindFirstChildWhichIsA("ProximityPrompt", true)
		if p then
			return p
		end
	end
	return nil
end

local function TriggerChuPrompt(prompt)
	if not prompt then
		return false
	end
	local ok = false
	pcall(function()
		prompt.HoldDuration = 0
		if fireproximityprompt then
			fireproximityprompt(prompt)
			ok = true
		end
	end)
	return ok
end

local function ReadChuDialogueFromGui()
	local pg = LocalPlayer:FindFirstChild("PlayerGui")
	local mg = pg and pg:FindFirstChild("MainGui")
	local menu = mg and mg:FindFirstChild("Menu")
	local dialogue = menu and menu:FindFirstChild("Dialogue")
	if dialogue and dialogue.Visible then
		for _, desc in ipairs(dialogue:GetDescendants()) do
			if desc:IsA("TextLabel") and desc.Visible then
				local txt = tostring(desc.Text or ""):gsub("^%s*(.-)%s*$", "%1")
				if
					#txt > 3
					and not string.find(txt, "Close")
					and not string.find(txt, "Main Quest")
					and txt ~= "???"
					and not string.find(txt, "Chu Tianjiao")
				then
					return txt
				end
			end
		end
	end
	return nil
end

local function CheckChuTianjiaoDialogue(timeoutSec)
	local char = LocalPlayer.Character or cachedChar
	local hrp = (char and char:FindFirstChild("HumanoidRootPart")) or cachedHrp
	if hrp then
		hrp.CFrame = CFrame.new(-2569.92, 15.74, -1405.49, 0.28, -0.00, 0.96, 0.00, 1.00, 0.00, -0.96, 0.00, 0.28)
	end
	task.wait(0.2)
	local startWait = os.clock()
	local limit = timeoutSec or 3.5
	local resultText = ""
	local lastText = ""
	local stableCount = 0
	while (os.clock() - startWait) < limit do
		local guiText = ReadChuDialogueFromGui()
		if not guiText or guiText == "" then
			local prompt = GetChuPrompt()
			if prompt then
				TriggerChuPrompt(prompt)
			end
		else
			resultText = guiText
			local lower = string.lower(guiText)
			if
				string.find(lower, "isle!", 1, true)
				or string.find(lower, "approaching", 1, true)
				or string.find(lower, "don't have any quests", 1, true)
				or string.find(lower, "maybe you'll find", 1, true)
			then
				break
			end
			if guiText == lastText and #guiText > 10 then
				stableCount = stableCount + 1
				if stableCount >= 3 then
					break
				end
			else
				lastText = guiText
				stableCount = 0
			end
		end
		task.wait(0.1)
	end
	return (resultText ~= ""), resultText
end

local function GetSacredBeastIslandCFrame(dialogueText)
	if not dialogueText or dialogueText == "" then
		return nil, "Unknown"
	end
	local lower = string.lower(dialogueText)
	for name, cf in pairs(IslandFarmPositions) do
		local cleanName = tostring(name):gsub("%s*|%s*%d+", ""):gsub("^%s+", ""):gsub("%s+$", "")
		if string.find(lower, string.lower(cleanName), 1, true) then
			return cf, cleanName
		end
	end
	local sp = workspace:FindFirstChild("Spawnpoint")
	if sp then
		for _, child in ipairs(sp:GetChildren()) do
			if string.find(lower, string.lower(child.Name), 1, true) then
				if IslandFarmPositions[child.Name] then
					return IslandFarmPositions[child.Name], child.Name
				end
				local p = child:IsA("BasePart") and child or child:FindFirstChildWhichIsA("BasePart", true)
				if p then
					return p.CFrame + Vector3.new(0, 5, 0), child.Name
				end
			end
		end
	end
	for name, cf in pairs(IslandFarmPositions) do
		local cleanName = tostring(name):gsub("%s*|%s*%d+", ""):gsub("^%s+", ""):gsub("%s+$", "")
		local prefix = cleanName:gsub("%s+[Ii]sle$", "")
		if #prefix >= 4 and string.find(lower, string.lower(prefix), 1, true) and string.find(lower, "isle", 1, true) then
			return cf, cleanName
		end
	end
	local matchedIsle = string.match(dialogueText, "([%u][%w%s]-%s+[Ii]sle)%s*!")
		or string.match(dialogueText, "([%w]+%s+[Ii]sle)")
	if matchedIsle then
		return nil, matchedIsle
	end
	return nil, "Unknown"
end

local function IsSacredBeastAppeared(dialogueText)
	local lower = string.lower(dialogueText or "")
	if lower == "" then
		return false
	end
	if
		string.find(lower, "i can feel it", 1, true)
		or string.find(lower, "approaching", 1, true)
		or string.find(lower, "don't have any quests", 1, true)
		or string.find(lower, "maybe you'll find", 1, true)
	then
		return false
	end
	if not string.find(lower, "hurry", 1, true) then
		return false
	end
	local _, islandName = GetSacredBeastIslandCFrame(dialogueText)
	if islandName and islandName ~= "Unknown" then
		return true
	end
	return string.find(lower, "isle!", 1, true) ~= nil
end

local function IsSpecialWeather(wName)
	local lower = string.lower(tostring(wName or ""))
	if lower == "" or lower == "clear" or lower == "sun" or lower == "sunny" or lower == "normal" then
		return false
	end
	if string.find(lower, "rain", 1, true)
		or string.find(lower, "thunder", 1, true)
		or string.find(lower, "storm", 1, true)
		or string.find(lower, "snow", 1, true)
		or string.find(lower, "frost", 1, true)
		or string.find(lower, "fog", 1, true)
		or string.find(lower, "mist", 1, true)
		or string.find(lower, "wind", 1, true)
		or string.find(lower, "eclipse", 1, true)
	then
		return true
	end
	return false
end

	end)
end

local function GetFishNameFromItem(itemName)
	if not itemName or itemName == "" then
		return ""
	end
	local cleanStr = string.split(tostring(itemName), "| Favorite")[1]
	cleanStr = string.split(cleanStr, "|")[1]
	cleanStr = string.gsub(cleanStr, "^%s*(.-)%s*$", "%1")
	local clientMod = ReplicatedStorage:FindFirstChild("ClientModule")
	local gs = clientMod and clientMod:FindFirstChild("GameService")
	if gs then
		local ok, res = pcall(function()
			local gMod = require(gs)
			return gMod and gMod.DestringFish and gMod.DestringFish(cleanStr)
		end)
		if ok and res and res.FishName then
			return res.FishName
		end
	end
	local parts = string.split(cleanStr, ",")
	if #parts >= 1 and parts[1] ~= "" then
		return parts[1]
	end
	return cleanStr
end

local function IsBossFish(fishName)
	local clean = GetFishNameFromItem(fishName)
	if clean == "" then
		return false
	end
	local inv = Info and Info:FindFirstChild("Inventory")
	if inv then
		local mod = inv:FindFirstChild(clean)
		if not mod then
			local lowerClean = string.lower(clean)
			for _, child in ipairs(inv:GetChildren()) do
				if string.lower(child.Name) == lowerClean then
					mod = child
					break
				end
			end
		end
		if mod then
			local data = GetModuleData(mod)
			if data and data.Boss == true then
				return true
			end
		end
	end
	if table.find(BossList, clean) then
		return true
	end
	local cleanLower = string.lower(clean)
	for _, b in ipairs(BossList) do
		if string.lower(b) == cleanLower then
			return true
		end
	end
	return false
end

local lastInfJumpTick = 0

local function IsInsideOceanWater(hrp)
	if not hrp then
		return false
	end
	local ocean = workspace:FindFirstChild("Ocean")
	if not ocean then
		return false
	end
	local pos = hrp.Position
	for _, child in ipairs(ocean:GetChildren()) do
		if child.Name == "Water" and child:IsA("BasePart") then
			local half = child.Size * 0.5
			local diff = child.Position - pos
			if math.abs(diff.X) <= half.X and math.abs(diff.Y) <= half.Y and math.abs(diff.Z) <= half.Z then
				return true
			end
		end
	end
	return false
end

local flyConnection = nil
local jumpStateConnection = nil

local function startFly()
	local char = LocalPlayer.Character
	if not char then
		return
	end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	local humanoid = char:FindFirstChildOfClass("Humanoid")

	if not hrp or not humanoid then
		return
	end

	humanoid.PlatformStand = true

	local bv = Instance.new("BodyVelocity")
	bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	bv.Velocity = Vector3.zero
	bv.Parent = hrp

	local bg = Instance.new("BodyGyro")
	bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	bg.P = 1e7
	bg.D = 12000
	bg.CFrame = hrp.CFrame
	bg.Parent = hrp

	jumpStateConnection = humanoid.StateChanged:Connect(function(_, new)
		if new == Enum.HumanoidStateType.Jumping or new == Enum.HumanoidStateType.Freefall then
			humanoid.PlatformStand = true
		end
	end)

	flyConnection = RunService.RenderStepped:Connect(function()
		local cam = workspace.CurrentCamera
		local move = Vector3.zero
		local md = humanoid.MoveDirection

		if md.Magnitude > 0 then
			local flatLook = Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z)
			local flatRight = Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z)

			if flatLook.Magnitude > 0.01 and flatRight.Magnitude > 0.01 then
				local fwdAmt = md:Dot(flatLook.Unit)
				local rightAmt = md:Dot(flatRight.Unit)

				move += cam.CFrame.LookVector * fwdAmt
				move += cam.CFrame.RightVector * rightAmt
			end
		end

		if move.Magnitude > 0 then
			move = move.Unit
		end
		bv.Velocity = move * Config.FlySpeed * 50
		local look = cam.CFrame.LookVector
		bg.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + look, Vector3.new(0, 1, 0))
	end)
end

local function stopFly()
	local char = LocalPlayer.Character
	if not char then
		return
	end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	local humanoid = char:FindFirstChildOfClass("Humanoid")

	if humanoid then
		humanoid.PlatformStand = false
	end
	if jumpStateConnection then
		jumpStateConnection:Disconnect()
		jumpStateConnection = nil
	end
	if flyConnection then
		flyConnection:Disconnect()
		flyConnection = nil
	end
	if hrp then
		for _, v in pairs(hrp:GetChildren()) do
			if v:IsA("BodyVelocity") or v:IsA("BodyGyro") then
				v:Destroy()
			end
		end
	end
end

		if not (skyPlatform and skyPlatform.Parent == workspace) then
			local old = workspace:FindFirstChild(SKY_PLATFORM_NAME)
			if old then

				end)

		local c = LocalPlayer.Character or cachedChar

	for _, entry in ipairs(inv:GetChildren()) do
		local rawName = entry.Name
		local cleanName = string.split(rawName, "| Favorite")[1]
		cleanName = string.gsub(cleanName, "^%s*(.-)%s*$", "%1")
		local matchedName = (rawName == fishName) or (cleanName == fishName)
		local weightNum = nil
		if gsMod and gsMod.DestringFish then
			local ok, res = pcall(gsMod.DestringFish, cleanName)
			if ok and res then
				if res.FishName and res.FishName == fishName then
					matchedName = true
				end
				if res.Weight then
					weightNum = tonumber(res.Weight)
				end
			end
		end
		if not weightNum then
			if entry:IsA("StringValue") then
				local parts = {}
				for p in string.gmatch(entry.Value, "[^|]+") do
					table.insert(parts, p)
				end
				if #parts >= 2 then
					weightNum = tonumber(parts[2]:gsub("^%s+", ""):gsub("%s+$", ""))
				elseif #parts == 1 then
					weightNum = tonumber(parts[1]:gsub("^%s+", ""):gsub("%s+$", ""))
				end
			elseif entry:IsA("NumberValue") or entry:IsA("IntValue") then
				weightNum = tonumber(entry.Value)
			end
		end
		if not weightNum then
			for num in string.gmatch(cleanName, "(%d+%.?%d*)%s*[kK][gG]") do
				weightNum = tonumber(num)
				break
			end
		end
		if matchedName and weightNum and weightNum >= minKg and weightNum <= maxKg then
			count = count + 1
		end
	end
	return count
end

local isMaterialFarmCastingAllowed = false

local function IsWeatherMatching(curWeather, weatherTable)
	if not weatherTable then
		return true
	end
	local lowerW = string.lower(tostring(curWeather or ""))
	if lowerW == "" then
		return false
	end
	for k, _ in pairs(weatherTable) do
		local lowerK = string.lower(tostring(k))
		if lowerW == lowerK or string.find(lowerW, lowerK, 1, true) or string.find(lowerK, lowerW, 1, true) then
			return true
		end
	end
	return false
end

local function IsGameAllowingSkill(char)
	if not FishingGui or not FishingGui.Visible then
		wasMinigameVisible = false
		return false
	end

		wasMinigameVisible = true
		minigameOpenTick = os.clock()
	end

	if (RhythmGui and RhythmGui.Visible) or (BossFightBar and BossFightBar.Visible) then
		return false
	end

	if char:GetAttribute("Phase2") == true then
		bossPhase2SetupUntil = 0
		return false
	end

	if bossPhase2SetupUntil > 0 and os.clock() < bossPhase2SetupUntil then
		return false
	end

	if char:GetAttribute("SkillLocked") or char:GetAttribute("Confused") then
		return false
	end

	if stats and stats:FindFirstChild("Stun") and stats.Stun.Value == true then
		return false
	end

	if fishObj then
		local readyVal = fishObj:FindFirstChild("Ready")
		if readyVal then
			if readyVal.Value ~= true then
				return false
			end
		elseif os.clock() - minigameOpenTick < 3.1 then
			return false
		end

		for _, s in ipairs(skillsFolder:GetChildren()) do
			if s:IsA("NumberValue") and (s:GetAttribute("UsingSkill") or 0) > 0 then
				return false
			end
		end
	end

local function IsSkillSlotAvailable(char, key)
	if PlayerData then

		local rodInv = PlayerData:FindFirstChild("FishingRodInventory")
		local rf = (rodVal and rodVal.Value ~= "" and rodInv) and rodInv:FindFirstChild(rodVal.Value) or nil
		local skFolder = rf and rf:FindFirstChild("Skill")
		local skVal = skFolder and skFolder:FindFirstChild(key)
		if skVal and skVal:IsA("StringValue") then
			return skVal.Value ~= ""
		end
	end
	local btn = FindSkillButton(key)

		return btn.Visible == true
	end
	local s = FindSkillNumberValue(char, key)

local function IsSkillOnCooldown(char, key)
	local btn = FindSkillButton(key)
	if btn then
		local cd = btn:FindFirstChild("CD")
		if cd and cd.Visible then
			local txt = cd.Text or ""
			local num = tonumber(string.match(txt, "%d+%.?%d*"))
			if num and num > 0 then
				return true
			end
		end
	end
	local s = FindSkillNumberValue(char, key)
	if s and (s.Value > 0 or (s:GetAttribute("UsingSkill") or 0) > 0) then
		return true
	end
	return false
end

local function IsSkillConfirmedUsed(char, key)
	return IsSkillOnCooldown(char, key)
end

local function IsSkillKeyReady(char, key)
	if not IsSkillSlotAvailable(char, key) then
		return false
	end

	if IsSkillOnCooldown(char, key) then
		return false
	end

	local btn = FindSkillButton(key)
	if btn then
		if not btn.Visible then
			return false
		end
		local lock = btn:FindFirstChild("Lock")
		if lock and lock.Visible then
			return false
		end
	end

	local s, skillsFolder = FindSkillNumberValue(char, key)
	if s then
		return true
	end

	if skillsFolder and #skillsFolder:GetChildren() > 0 then
		return false
	end

	return btn ~= nil and btn.Visible == true
end

local getupval = (debug and debug.getupvalue) or getupvalue

local function StepPhase2MiniGame()
	if not BossFightBar or not BossFightBar.Visible then
		if #phase2BarHistory > 0 then
			table.clear(phase2BarHistory)
		end
		return
	end

	local bar = BossFightBar:FindFirstChild("Bar")
	local hitbox = BossFightBar:FindFirstChild("Hitbox")
	if not bar or not hitbox then
		return
	end

	local barW = bar.Size.X.Scale
	local barCenter = bar.Position.X.Scale + barW * (0.5 - bar.AnchorPoint.X)
	local hbW = hitbox.Size.X.Scale
	local hbLeft = hitbox.Position.X.Scale - hbW * hitbox.AnchorPoint.X

	local now = os.clock()
	table.insert(phase2BarHistory, { t = now, x = barCenter })
	while #phase2BarHistory > 1 and (now - phase2BarHistory[1].t) > 0.15 do
		table.remove(phase2BarHistory, 1)
	end

	local hitDelay = (tonumber(Config.BossHitDelay) or 15) / 1000
	if hitDelay < 0.005 then
		hitDelay = 0.005
	end

	local handledViaUpval = false
	local onClickFn = nil

	if getconnections and getupval and MobileFishingBtn then
		pcall(function()
			for _, conn in pairs(getconnections(MobileFishingBtn.MouseButton1Down)) do
				local fn = conn.Function and getupval(conn.Function, 1)
				if type(fn) == "function" then
					onClickFn = fn
					local u12 = getupval(fn, 1)
					local u14 = getupval(fn, 2)
					local u27 = getupval(fn, 3)
					local u26 = getupval(fn, 4)
					local u1 = getupval(fn, 5)
					local u25 = getupval(fn, 6)
					local u23 = getupval(fn, 7)
					local u2 = getupval(fn, 8)
					local u15 = getupval(fn, 9)
					if
						u12 == true
						and type(u14) == "number"
						and type(u27) == "number"
						and type(u26) == "number"
						and type(u1) == "number"
						and type(u25) == "number"
						and type(u23) == "number"
						and type(u2) == "number"
						and type(u15) == "number"
					then
						if math.abs(u2 - hbW) < 0.005 and math.abs(u15 - hbLeft) < 0.005 then
							handledViaUpval = true
							if (now - u14 >= hitDelay or u14 == 0) and now - lastPhase2HitTick >= hitDelay then
								local v91 = (now - 0.025 - u27) % u26 * u1
								if u25 < v91 then
									v91 = u25 * 2 - v91
								end
								local v92 = u23 + v91
								local v93 = u2 * 0.3
								if v92 >= (u15 - v93 * 0.8) and v92 <= (u15 + u2 + v93 * 0.8) then
									lastPhase2HitTick = now
									fn(false)
								end
							end
						end
						break
					end
				end
			end
		end)
	end

	if handledViaUpval then
		return
	end

	if now - lastPhase2HitTick < hitDelay then
		return
	end

	local evalX = barCenter
	local pad = hbW * 0.2

	local badBox = BossFightBar:FindFirstChild("HitboxBad")
	if badBox and badBox.Visible then
		local badW = badBox.Size.X.Scale
		local badLeft = badBox.Position.X.Scale - badW * badBox.AnchorPoint.X
		if evalX >= (badLeft - 0.01) and evalX <= (badLeft + badW + 0.01) then
			return
		end
	end

	if evalX >= (hbLeft - pad) and evalX <= (hbLeft + hbW + pad) then
		lastPhase2HitTick = now
		local clicked = false
		if onClickFn then
			pcall(function()
				onClickFn(false)
				clicked = true
			end)
		end
		if not clicked and MobileFishingBtn then
			if getconnections then
				pcall(function()
					for _, conn in pairs(getconnections(MobileFishingBtn.MouseButton1Down)) do
						if conn.Function then
							task.spawn(conn.Function)
							clicked = true
						elseif conn.Fire then
							conn:Fire()
							clicked = true
						end
					end
				end)
			end
			if not clicked and firesignal then
				pcall(function()
					firesignal(MobileFishingBtn.MouseButton1Down)
					clicked = true
				end)
			end
		end
		if not clicked then
			pcall(function()
				VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 0)
				task.delay(0.005, function()
					VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0)
				end)
			end)
		end
	end
end

local function IsTrapNote(gui)
	if not gui or not gui:IsA("GuiObject") then
		return false
	end
	if gui:GetAttribute("Trap") == true or gui:GetAttribute("trap") == true or gui:GetAttribute("IsTrap") == true then
		return true
	end
	if string.find(string.lower(gui.Name), "trap", 1, true) then
		return true
	end
	local function isRed(c)
		return c and c.R > 0.65 and c.G < 0.45 and c.B < 0.45 and c.R > c.G * 1.8 and c.R > c.B * 1.8
	end
	if isRed(gui.BackgroundColor3) then
		return true
	end
	if (gui:IsA("ImageLabel") or gui:IsA("ImageButton")) and isRed(gui.ImageColor3) then
		return true
	end
	for _, desc in ipairs(gui:GetDescendants()) do
		if desc:IsA("GuiObject") then
			if isRed(desc.BackgroundColor3) then
				return true
			end
			if (desc:IsA("ImageLabel") or desc:IsA("ImageButton")) and isRed(desc.ImageColor3) then
				return true
			end
		end
	end
	return false
end

local function StepRhythmMiniGame()
	if not RhythmGui or not RhythmGui.Visible then
		return
	end
	local now = os.clock()
	if now - lastRhythmScan < 0.025 then
		return
	end
	lastRhythmScan = now

	for _, lane in ipairs(rhythmLanes) do
		if now - (lastRhythmLaneHit[lane] or 0) >= 0.05 then
			local laneFrame = RhythmGui:FindFirstChild("Progression" .. lane)
			if laneFrame then
				local barFrame = laneFrame:FindFirstChild("BarFrame")
				local targetY = barFrame and barFrame.Position.Y.Scale or 0.85
				local closestNote = nil
				local closestDist = 999
				for _, child in ipairs(laneFrame:GetChildren()) do
					if child.Name == "Note_FX" and child:IsA("GuiObject") and child.Visible then
						local dist = math.abs(child.Position.Y.Scale - targetY)
						if dist < closestDist then
							closestDist = dist
							closestNote = child
						end
					end
				end
				if closestNote and not IsTrapNote(closestNote) and closestDist <= 0.18 then
					lastRhythmLaneHit[lane] = now
					local btn = laneFrame:FindFirstChild("Button")
					if btn then
						TriggerGuiButton(btn)
					end
					if closestNote.Parent ~= nil then
						local kc = rhythmKeyMap[lane]
						if kc then
							pcall(function()
								VirtualInputManager:SendKeyEvent(true, kc, false, game)
								task.delay(0.02, function()
									VirtualInputManager:SendKeyEvent(false, kc, false, game)
								end)
							end)
						end
					end
					if closestNote.Parent ~= nil and closestDist <= 0.1 and RhythmHitEvent then
						pcall(function()
							closestNote:Destroy()
							RhythmHitEvent:FireServer("hit")
						end)
					end
				end
			end
		end
	end
end

local function GetModelPart(obj)
	if not obj then
		return nil
	end
	if obj:IsA("BasePart") then
		return obj
	end
	if obj:IsA("Model") then
		return obj:FindFirstChild("HumanoidRootPart")
			or obj:FindFirstChild("Head")
			or obj.PrimaryPart
			or obj:FindFirstChildWhichIsA("BasePart", true)
	end
	return obj:FindFirstChildWhichIsA("BasePart", true)
end

local function IsSecretBossEntity(obj)
	if not obj then
		return false
	end
	if obj:GetAttribute("Secret") == true or obj:GetAttribute("Rarity") == "Secret" then
		return true
	end
	local rawName = tostring(obj:GetAttribute("Name") or obj:GetAttribute("FishName") or obj.Name)
	local lowerName = string.lower(rawName)
	if
		string.find(lowerName, "secret")
		or string.find(lowerName, "long diaohai")
		or string.find(lowerName, "longdiaohai")
		or string.find(lowerName, "diaohai")
		or string.find(lowerName, "wuzhi")
	then
		return true
	end
	local inv = Info:FindFirstChild("Inventory")
	local mod = inv and inv:FindFirstChild(rawName)
	local data = GetModuleData(mod)
	if
		data
		and (
			data.Secret == true
			or data.Rarity == "Secret"
			or (data.Boss == true and (tonumber(data.Chance) or 100) <= 1)
		)
	then
		return true
	end
	return false
end

local function FindBossTarget(wantSecret)
	local npcFolder = workspace:FindFirstChild("NPC")
	local bossFolder = npcFolder and npcFolder:FindFirstChild("Boss")

	if bossFolder then
		for _, boss in ipairs(bossFolder:GetChildren()) do
			local part = GetModelPart(boss)
			if part then
				local isSec = IsSecretBossEntity(boss)
				if wantSecret and isSec then
					return boss, part
				elseif not wantSecret and not isSec then
					return boss, part
				end
			end
		end
	end

	if not wantSecret then
		local wsBoss = workspace:FindFirstChild("Boss") or workspace:FindFirstChild("BossZone")
		if wsBoss then
			local part = GetModelPart(wsBoss)
			if part then
				return wsBoss, part
			end
		end
	end

	return nil, nil
end

local function ParseBossCooldownSeconds(text)
	if not text or text == "" then
		return nil
	end
	local lower = string.lower(text)
	local m, s = string.match(lower, "(%d+)%s*:%s*(%d+)")
	if m and s then
		return (tonumber(m) or 0) * 60 + (tonumber(s) or 0)
	end
	local mPart = string.match(lower, "(%d+)%s*m")
	local sPart = string.match(lower, "(%d+)%s*s")
	if mPart or sPart then
		return (tonumber(mPart) or 0) * 60 + (tonumber(sPart) or 0)
	end
	local secOnly = string.match(lower, "(%d+)%s*sec")
	if secOnly then
		return tonumber(secOnly) or 0
	end
	return nil
end

local function CheckBossCooldownText(text)
	if not text or text == "" then
		return
	end
	local lower = string.lower(text)
	local isEnzo = string.find(lower, "enzo", 1, true) ~= nil
	local isLong = (string.find(lower, "long diaohai", 1, true) ~= nil)
		or (string.find(lower, "longdiaohai", 1, true) ~= nil)
		or (string.find(lower, "diaohai", 1, true) ~= nil)

	if not isEnzo and not isLong then
		return
	end

	local now = os.clock()
	if string.find(lower, "cooldown", 1, true) then
		local sec = ParseBossCooldownSeconds(lower)
		if sec and sec > 0 and sec <= 3600 then
			if isEnzo then
				enzoCooldownUntil = now + sec
			elseif isLong then
				longDiaohaiCooldownUntil = now + sec
			end
		end
	elseif string.find(lower, "already being fought", 1, true) then
		if isEnzo then
			if now >= enzoCooldownUntil then
				enzoCooldownUntil = now + 5
			end
		elseif isLong then
			if now >= longDiaohaiCooldownUntil then
				longDiaohaiCooldownUntil = now + 5
			end
		end
	end
end

local function StepAutoBoss(char, isMinigame)
	if isMinigame then
		return
	end

	local enzoOn = Config.AutoBoss == true
	local longOn = Config.AutoSecretBoss == true

	if not enzoOn and not longOn then
		return
	end

	local now = os.clock()
	if now - lastBossActionTick < 0.5 then
		return
	end
	lastBossActionTick = now

	local enzoReady = enzoOn and (now >= enzoCooldownUntil)
	local longReady = longOn and (now >= longDiaohaiCooldownUntil)

	local targetBoss = nil
	if enzoOn and longOn then
		if enzoReady and not longReady then
			targetBoss = "Enzo"
		elseif longReady and not enzoReady then
			targetBoss = "Long Diaohai"
		elseif enzoReady and longReady then
			if currentBossTarget == "Long Diaohai" then
				targetBoss = "Long Diaohai"
			else
				targetBoss = "Enzo"
			end
		else
			if enzoCooldownUntil <= longDiaohaiCooldownUntil then
				targetBoss = "Enzo"
			else
				targetBoss = "Long Diaohai"
			end
		end
	elseif enzoOn then
		targetBoss = "Enzo"
	elseif longOn then
		targetBoss = "Long Diaohai"
	end

	if not targetBoss then
		return
	end

	currentBossTarget = targetBoss

	local isTargetReady = (targetBoss == "Long Diaohai") and (now >= longDiaohaiCooldownUntil) or (now >= enzoCooldownUntil)

	local c = char or cachedChar or LocalPlayer.Character
	if c and c:GetAttribute("Type") ~= "Fishing Rod" then
		pcall(function()
			ToggleHotbar:InvokeServer("1")
		end)
	end

	if isTargetReady then
		pcall(function()
			local args = {
				[1] = targetBoss,
				[2] = Config.BossMode or "Normal",
			}
			game:GetService("ReplicatedStorage"):WaitForChild("Events", 9e9):WaitForChild("StartBossFight", 9e9):FireServer(unpack(args))
		end)
	end
end

		end
		if desc:IsA("PVInstance") then
			local ok, cf = pcall(function()
				return desc:GetPivot()
			end)
			if ok and cf and cf.Position.Magnitude > 1 then
				return cf.Position, cf, nil, desc
			end
		end
		return nil, nil, nil, nil
	end

	local function checkObj(desc)
		if not desc or desc:IsA("Folder") then
			return
		end
		if not (desc:IsA("Model") or desc:IsA("BasePart")) then
			return
		end
		local dName = tostring(desc.Name or "")
		local aName = tostring(desc:GetAttribute("Name") or "")
		if dName == "God" or dName == "Lamp" or aName == "God" or aName == "Lamp" then
			return
		end
		local dLower = string.lower(string.gsub(dName, "%s+", ""))
		local aLower = string.lower(string.gsub(aName, "%s+", ""))

		local col = nil
		if string.find(dLower, "yellow", 1, true) or string.find(aLower, "yellow", 1, true) then
			col = "yellow"
		elseif string.find(dLower, "green", 1, true) or string.find(aLower, "green", 1, true) then
			col = "green"
		elseif string.find(dLower, "blue", 1, true) or string.find(aLower, "blue", 1, true) then
			col = "blue"
		end

		local isInsideGod = false
		local p = desc.Parent
		while p and p ~= workspace do
			if p.Name == "God" then
				isInsideGod = true
				break
			end
			p = p.Parent
		end

		if not col and isInsideGod then
			col = dLower ~= "" and dLower or "yellow"
		end

		if col and not seen[col] then
			local pos, cf, part, obj = resolveEntityTarget(desc)
			if pos then
				seen[col] = true
				table.insert(results, {
					name = (dName ~= "" and dName ~= "God") and dName or col,
					part = part,
					obj = obj,
					pos = pos,
					cf = cf,
					lower = col,
				})
			end
		end
	end

	local npcGod = workspace:FindFirstChild("NPC") and workspace.NPC:FindFirstChild("God")
	if npcGod then
		for _, child in ipairs(npcGod:GetChildren()) do
			checkObj(child)
		end
		if #results > 0 then
			return results
		end
	end

	local godFolder = workspace:FindFirstChild("God")
	if godFolder then
		for _, child in ipairs(godFolder:GetChildren()) do
			checkObj(child)
		end
		if #results > 0 then
			return results
		end
	end

	local gsFolder = workspace:FindFirstChild("GodSpirit")
	if gsFolder then
		for _, child in ipairs(gsFolder:GetChildren()) do
			checkObj(child)
		end
		if #results > 0 then
			return results
		end
	end

	return results
end

local function FindSpecialEntities()
	local candidatesMaoshan = {}
	local candidatesTaoist = {}

	local function checkCandidate(desc, folderName)
		if not desc or desc:IsA("Folder") then
			return
		end
		if not (desc:IsA("Model") or desc:IsA("BasePart")) then
			return
		end
		local dName = tostring(desc.Name or "")
		local aName = tostring(desc:GetAttribute("Name") or "")
		if dName == "God" or dName == "Lamp" or (dName == "" and aName == "") then
			return
		end
		local part = GetModelPart(desc)
		local pos = nil
		local cf = nil
		if part and part:IsDescendantOf(workspace) then
			pos = part.Position
			cf = part.CFrame
		elseif desc:IsA("PVInstance") then
			local ok, pcf = pcall(function()
				return desc:GetPivot()
			end)
			if ok and pcf and pcf.Position.Magnitude > 1 then
				pos = pcf.Position
				cf = pcf
			end
		end
		if not pos then
			return
		end

		local dLower = string.lower(string.gsub(dName, "%s+", ""))
		local aLower = string.lower(string.gsub(aName, "%s+", ""))

		local isMao = string.find(dLower, "maoshan", 1, true)
			or string.find(dLower, "maosan", 1, true)
			or string.find(aLower, "maoshan", 1, true)
			or string.find(aLower, "maosan", 1, true)

		local isTao = string.find(dLower, "taoist", 1, true)
			or string.find(dLower, "taosi", 1, true)
			or string.find(aLower, "taoist", 1, true)
			or string.find(aLower, "taosi", 1, true)

		if not isMao and not isTao then
			for _, attrVal in pairs(desc:GetAttributes()) do
				local valLower = string.lower(string.gsub(tostring(attrVal), "%s+", ""))
				if string.find(valLower, "maoshan", 1, true) or string.find(valLower, "maosan", 1, true) then
					isMao = true
					break
				elseif string.find(valLower, "taoist", 1, true) or string.find(valLower, "taosi", 1, true) then
					isTao = true
					break
				end
			end
		end

		if isMao then
			table.insert(candidatesMaoshan, { part = part, pos = pos, cf = cf, folder = folderName, obj = desc })
		elseif isTao then
			table.insert(candidatesTaoist, { part = part, pos = pos, cf = cf, folder = folderName, obj = desc })
		end
	end

	local wsNPC = workspace:FindFirstChild("NPC")
	if wsNPC then
		for _, desc in ipairs(wsNPC:GetDescendants()) do
			checkCandidate(desc, "NPC")
		end
	end

	local finalMaoshan = candidatesMaoshan[1] or nil
	local finalTaoist = candidatesTaoist[1] or nil

	local godSpirits = FindAllGodSpirits()
	return finalMaoshan, finalTaoist, godSpirits
end

local ActiveEntityESPs = {}

local function destroyEntityESPEntry(entry)
	if not entry then
		return
	end
	if type(entry) == "userdata" then
		pcall(function()
			entry:Destroy()
		end)
		return
	end

		pcall(function()
			entry.gui:Destroy()
		end)

local function applyEntityESP(targetInfo, tag, title)
	if not targetInfo or not targetInfo.pos then
		return
	end
	local entityFill = Color3.fromRGB(170, 50, 255)
	local entityOutline = Color3.fromRGB(225, 165, 255)
	local entityText = Color3.fromRGB(195, 100, 255)
	if tag == "Taoist" then
		entityFill = Color3.fromRGB(255, 215, 0)
		entityOutline = Color3.fromRGB(255, 240, 150)
		entityText = Color3.fromRGB(255, 225, 50)
	elseif tag == "Maoshan" then
		entityFill = Color3.fromRGB(115, 118, 200)
		entityOutline = Color3.fromRGB(165, 170, 255)
		entityText = Color3.fromRGB(140, 145, 240)
	elseif string.find(tag, "blue") then
		entityFill = Color3.fromRGB(0, 150, 255)
		entityOutline = Color3.fromRGB(120, 210, 255)
		entityText = Color3.fromRGB(60, 180, 255)
	elseif string.find(tag, "yellow") then
		entityFill = Color3.fromRGB(255, 230, 0)
		entityOutline = Color3.fromRGB(255, 255, 150)
		entityText = Color3.fromRGB(255, 240, 50)
	elseif string.find(tag, "green") then
		entityFill = Color3.fromRGB(0, 230, 80)
		entityOutline = Color3.fromRGB(140, 255, 170)
		entityText = Color3.fromRGB(60, 255, 110)
	end
	local part = targetInfo.part
	local obj = targetInfo.obj or part
	local pos = targetInfo.pos

	local entry = ActiveEntityESPs[tag]
	if not entry or type(entry) ~= "table" then
		if entry then
			destroyEntityESPEntry(entry)
		end
		entry = {}
		ActiveEntityESPs[tag] = entry
	end

	local adornee = nil
	if part and part:IsDescendantOf(workspace) then
		adornee = part
		if entry.att then
			pcall(function()
				entry.att:Destroy()
			end)
			entry.att = nil
		end
	else
		local terrain = workspace:FindFirstChildOfClass("Terrain")
		if terrain then
			if not (entry.att and entry.att.Parent == terrain) then
				local att = Instance.new("Attachment")
				att.Name = "namdzEntityAtt_" .. tag
				att.Parent = terrain
				entry.att = att
			end
			entry.att.WorldPosition = pos
			adornee = entry.att
		end
	end

	if entry.hl then
		pcall(function()
			entry.hl:Destroy()
		end)
		entry.hl = nil
	end

	if adornee then
		if not (entry.gui and entry.gui.Parent) then
			local esp = Instance.new("BillboardGui")
			esp.Name = "namdzEntityESP"
			esp.Size = UDim2.new(0, 220, 0, 26)
			esp.StudsOffset = Vector3.new(0, 3.5, 0)
			esp.AlwaysOnTop = true
			esp.MaxDistance = math.huge
			local text = Instance.new("TextLabel")
			text.Name = "NameText"
			text.Parent = esp
			text.Size = UDim2.new(1, 0, 1, 0)
			text.BackgroundTransparency = 1
			text.TextScaled = false
			text.TextSize = 13
			text.Font = Enum.Font.GothamBold
			text.TextColor3 = entityText
			text.TextStrokeTransparency = 0
			text.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
			text.TextXAlignment = Enum.TextXAlignment.Center
			entry.gui = esp
		end
		entry.gui.Adornee = adornee
		if entry.gui.Parent ~= adornee then
			entry.gui.Parent = adornee
		end
		local nameLabel = entry.gui:FindFirstChild("NameText")
		if nameLabel then
			nameLabel.TextColor3 = entityText
			nameLabel.Text = title
		end
	end
end

local function clearEntityESP(tag)
	if tag then
		if tag == "GodSpirit" then
			for k, entry in pairs(ActiveEntityESPs) do
				if string.find(k, "GodSpirit_") then
					destroyEntityESPEntry(entry)
					ActiveEntityESPs[k] = nil
				end
			end
		else
			if ActiveEntityESPs[tag] then
				destroyEntityESPEntry(ActiveEntityESPs[tag])
				ActiveEntityESPs[tag] = nil
			end
		end
	else
		for k, entry in pairs(ActiveEntityESPs) do
			destroyEntityESPEntry(entry)
			ActiveEntityESPs[k] = nil
		end
	end
end

local function updateEntityESP(finalMaoshan, finalTaoist, godSpirits)
	if Config.ESPMaoshan and finalMaoshan then
		applyEntityESP(finalMaoshan, "Maoshan", "Maoshan")
	else
		clearEntityESP("Maoshan")
	end

	if Config.ESPTaoist and finalTaoist then
		applyEntityESP(finalTaoist, "Taoist", "Taoist")
	else
		clearEntityESP("Taoist")
	end

	if Config.ESPGodSpirit and #godSpirits > 0 then
		local currentKeys = {}
		for _, g in ipairs(godSpirits) do
			local tag = "GodSpirit_" .. g.lower
			currentKeys[tag] = true
			local dispName = g.lower == "yellow" and "Yellow"
				or (g.lower == "green" and "Green" or (g.lower == "blue" and "Blue" or (string.upper(string.sub(g.lower, 1, 1)) .. string.sub(g.lower, 2))))
			applyEntityESP(g, tag, dispName)
		end
		for k, _ in pairs(ActiveEntityESPs) do
			if string.find(k, "GodSpirit_") and not currentKeys[k] then
				clearEntityESP(k)
			end
		end
	else
		for k, _ in pairs(ActiveEntityESPs) do
			if string.find(k, "GodSpirit_") then
				clearEntityESP(k)
			end
		end
	end
end

local function SendDiscordWebhook(rawUrl, title, description, customColor, extraFields)
	local url = tostring(rawUrl or ""):gsub("%s+", "")
	if url == "" then
		return false
	end
	if not (string.find(url, "discord.com/api/webhooks") or string.find(url, "discordapp.com/api/webhooks") or string.find(url, "webhook")) then
		return false
	end
	local req = (syn and syn.request) or (http and http.request) or http_request or (fluxus and fluxus.request) or request
	if not req then
		return false
	end
	local fields = {}
	if type(extraFields) == "table" then
		for _, f in ipairs(extraFields) do
			table.insert(fields, f)
		end
	end
	table.insert(fields, {
		name = "Server JobId",
		value = tostring(game.JobId),
		inline = false,
	})
	local payload = {
		embeds = {
			{
				title = title,
				description = description,
				color = customColor or 65280,
				fields = fields,
				footer = {
					text = "StarCrow Hub",
				},
				timestamp = DateTime.now():ToIsoDate(),
			},
		},
	}
	local ok, res = pcall(function()
		return req({
			Url = url,
			Method = "POST",
			Headers = {
				["Content-Type"] = "application/json",
			},
			Body = HttpService:JSONEncode(payload),
		})
	end)
	if ok and res then
		local sc = res.StatusCode or res.status_code or res.Status
		if sc and tonumber(sc) and tonumber(sc) >= 200 and tonumber(sc) < 300 then
			return true
		end
		if not sc then
			return true
		end
	end
	return false
end

local function SendBossWebhook(title, description, customColor, islandName)
	local targetUrl = Config.WebhookUrl ~= "" and Config.WebhookUrl or Config.BossWebhookUrl

		{
			{
				name = "Island",
				value = tostring((islandName and islandName ~= "") and islandName or "Unknown"),
				inline = true,
			},
		}
	)
end

local function GetNearestIslandName(pos)
	if not pos then
		return "Unknown"
	end
	local bestName = "Unknown"
	local bestDist = math.huge
	for name, cf in pairs(IslandFarmPositions) do
		local d = (cf.Position - pos).Magnitude

local function CheckChuTianjiaoDialogue(timeoutSec)
	local char = LocalPlayer.Character or cachedChar
	local hrp = (char and char:FindFirstChild("HumanoidRootPart")) or cachedHrp
	if hrp then
		hrp.CFrame = CFrame.new(-2569.92, 15.74, -1405.49, 0.28, -0.00, 0.96, 0.00, 1.00, 0.00, -0.96, 0.00, 0.28)
	end
	task.wait(0.2)
	local startWait = os.clock()
	local limit = timeoutSec or 3.5
	local resultText = ""
	local lastText = ""
	local stableCount = 0
	while (os.clock() - startWait) < limit do
		local guiText = ReadChuDialogueFromGui()
		if not guiText or guiText == "" then
			local prompt = GetChuPrompt()
			if prompt then
				TriggerChuPrompt(prompt)
			end
		else
			resultText = guiText
			local lower = string.lower(guiText)
			if
				string.find(lower, "isle!", 1, true)
				or string.find(lower, "approaching", 1, true)
				or string.find(lower, "don't have any quests", 1, true)
				or string.find(lower, "maybe you'll find", 1, true)
			then
				break
			end
			if guiText == lastText and #guiText > 10 then
				stableCount = stableCount + 1
				if stableCount >= 3 then
					break
				end
			else
				lastText = guiText
				stableCount = 0
			end
		end
		task.wait(0.1)
	end
	return (resultText ~= ""), resultText
end

local function GetSacredBeastIslandCFrame(dialogueText)
	if not dialogueText or dialogueText == "" then
		return nil, "Unknown"
	end
	local lower = string.lower(dialogueText)
	for name, cf in pairs(IslandFarmPositions) do
		local cleanName = tostring(name):gsub("%s*|%s*%d+", ""):gsub("^%s+", ""):gsub("%s+$", "")
		if string.find(lower, string.lower(cleanName), 1, true) then
			return cf, cleanName
		end
	end
	local sp = workspace:FindFirstChild("Spawnpoint")
	if sp then
		for _, child in ipairs(sp:GetChildren()) do
			if string.find(lower, string.lower(child.Name), 1, true) then
				if IslandFarmPositions[child.Name] then
					return IslandFarmPositions[child.Name], child.Name
				end
				local p = child:IsA("BasePart") and child or child:FindFirstChildWhichIsA("BasePart", true)
				if p then
					return p.CFrame + Vector3.new(0, 5, 0), child.Name
				end
			end
		end
	end
	for name, cf in pairs(IslandFarmPositions) do
		local cleanName = tostring(name):gsub("%s*|%s*%d+", ""):gsub("^%s+", ""):gsub("%s+$", "")
		local prefix = cleanName:gsub("%s+[Ii]sle$", "")
		if #prefix >= 4 and string.find(lower, string.lower(prefix), 1, true) and string.find(lower, "isle", 1, true) then
			return cf, cleanName
		end
	end
	local matchedIsle = string.match(dialogueText, "([%u][%w%s]-%s+[Ii]sle)%s*!")
		or string.match(dialogueText, "([%w]+%s+[Ii]sle)")
	if matchedIsle then
		return nil, matchedIsle
	end
	return nil, "Unknown"
end

end

local function GetModuleData(mod)
	if not mod or not mod:IsA("ModuleScript") then
		return nil
	end
	local cached = ModuleCache[mod]
	if cached ~= nil then
		return cached ~= false and cached or nil
	end
	local ok, data = pcall(require, mod)
	if ok and type(data) == "table" then
		ModuleCache[mod] = data
		return data
	end
	ModuleCache[mod] = false
	return nil
end

local function GetFishNameFromItem(itemName)
	if not itemName or itemName == "" then
		return ""
	end
	local cleanStr = string.split(tostring(itemName), "| Favorite")[1]
	cleanStr = string.split(cleanStr, "|")[1]
	cleanStr = string.gsub(cleanStr, "^%s*(.-)%s*$", "%1")
	local clientMod = ReplicatedStorage:FindFirstChild("ClientModule")
	local gs = clientMod and clientMod:FindFirstChild("GameService")
	if gs then
		local ok, res = pcall(function()
			local gMod = require(gs)
			return gMod and gMod.DestringFish and gMod.DestringFish(cleanStr)
		end)
		if ok and res and res.FishName then
			return res.FishName
		end
	end
	local parts = string.split(cleanStr, ",")
	if #parts >= 1 and parts[1] ~= "" then
		return parts[1]
	end
	return cleanStr
end

local function IsBossFish(fishName)
	local clean = GetFishNameFromItem(fishName)
	if clean == "" then
		return false
	end
	local inv = Info and Info:FindFirstChild("Inventory")
	if inv then
		local mod = inv:FindFirstChild(clean)
		if not mod then
			local lowerClean = string.lower(clean)
			for _, child in ipairs(inv:GetChildren()) do
				if string.lower(child.Name) == lowerClean then
					mod = child
					break
				end
			end
		end
		if mod then
			local data = GetModuleData(mod)
			if data and data.Boss == true then
				return true
			end
		end
	end
	if table.find(BossList, clean) then
		return true
	end
	local cleanLower = string.lower(clean)
	for _, b in ipairs(BossList) do
		if string.lower(b) == cleanLower then
			return true
		end
	end
	return false
end

local lastInfJumpTick = 0

local function IsInsideOceanWater(hrp)
	if not hrp then
		return false
	end
	local ocean = workspace:FindFirstChild("Ocean")
	if not ocean then
		return false
	end
	local pos = hrp.Position
	for _, child in ipairs(ocean:GetChildren()) do
		if child.Name == "Water" and child:IsA("BasePart") then
			local half = child.Size * 0.5
			local diff = child.Position - pos
			if math.abs(diff.X) <= half.X and math.abs(diff.Y) <= half.Y and math.abs(diff.Z) <= half.Z then
				return true
			end
		end
	end
	return false
end

local flyConnection = nil

local function startFly()
	local char = LocalPlayer.Character
	if not char then
		return
	end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	local humanoid = char:FindFirstChildOfClass("Humanoid")

	if not hrp or not humanoid then
		return
	end

	humanoid.PlatformStand = true

	local bv = Instance.new("BodyVelocity")
	bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	bv.Velocity = Vector3.zero
	bv.Parent = hrp

	local bg = Instance.new("BodyGyro")
	bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	bg.P = 1e7
	bg.D = 12000
	bg.CFrame = hrp.CFrame
	bg.Parent = hrp

	jumpStateConnection = humanoid.StateChanged:Connect(function(_, new)
		if new == Enum.HumanoidStateType.Jumping or new == Enum.HumanoidStateType.Freefall then
			humanoid.PlatformStand = true
		end
	end)

	flyConnection = RunService.RenderStepped:Connect(function()
		local cam = workspace.CurrentCamera
		local move = Vector3.zero
		local md = humanoid.MoveDirection

		if md.Magnitude > 0 then
			local flatLook = Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z)
			local flatRight = Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z)

			if flatLook.Magnitude > 0.01 and flatRight.Magnitude > 0.01 then
				local fwdAmt = md:Dot(flatLook.Unit)
				local rightAmt = md:Dot(flatRight.Unit)

				move += cam.CFrame.LookVector * fwdAmt
				move += cam.CFrame.RightVector * rightAmt
			end
		end

		if move.Magnitude > 0 then
			move = move.Unit
		end
		bv.Velocity = move * Config.FlySpeed * 50
		local look = cam.CFrame.LookVector
		bg.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + look, Vector3.new(0, 1, 0))
	end)
end

local function stopFly()
	local char = LocalPlayer.Character
	if not char then
		return
	end

	local hrp = char:FindFirstChild("HumanoidRootPart")
	local humanoid = char:FindFirstChildOfClass("Humanoid")

	if humanoid then
		humanoid.PlatformStand = false
	end
	if jumpStateConnection then
		jumpStateConnection:Disconnect()
		jumpStateConnection = nil
	end
	if flyConnection then
		flyConnection:Disconnect()
		flyConnection = nil
	end
	if hrp then
		for _, v in pairs(hrp:GetChildren()) do
			if v:IsA("BodyVelocity") or v:IsA("BodyGyro") then
				v:Destroy()
			end
		end
	end
end

		if skyHeartbeatConn then

		end
		if skyPlatform then

		end
	end

	local function EnsureSkyFarmPlatform(x, h, z)
		if not (skyPlatform and skyPlatform.Parent == workspace) then
			local old = workspace:FindFirstChild(SKY_PLATFORM_NAME)
			if old then
				pcall(function()
					old:Destroy()
				end)
			end
			local p = Instance.new("Part")
			p.Name = SKY_PLATFORM_NAME
			p.Anchored = true
			p.CanCollide = true
			p.CanTouch = false

		local c = LocalPlayer.Character or cachedChar
		local hrp = (c and c:FindFirstChild("HumanoidRootPart")) or cachedHrp

			return
		end
		local targetX = (useSavedAnchor and skyAnchorX) or hrp.Position.X
		local targetZ = (useSavedAnchor and skyAnchorZ) or hrp.Position.Z
		local rot = (useSavedAnchor and skyAnchorRot) or hrp.CFrame.Rotation
		skyAnchorX = targetX
		skyAnchorZ = targetZ
		skyAnchorRot = rot
		EnsureSkyFarmPlatform(targetX, h, targetZ)
		pcall(function()
			hrp.AssemblyLinearVelocity = Vector3.zero
			hrp.AssemblyAngularVelocity = Vector3.zero
		end)
		hrp.CFrame = CFrame.new(targetX, h, targetZ) * rot
	end

		Config.SkyFarm = enabled == true
		if not Config.SkyFarm then
			DestroySkyPlatform()
			local prevY = skySavedY
			local prevX = skyAnchorX
			local prevZ = skyAnchorZ
			skySavedY = nil
			skyAnchorX = nil
			skyAnchorZ = nil
			skyAnchorRot = nil

				end)
				hrp.CFrame = CFrame.new(prevX or hrp.Position.X, prevY, prevZ or hrp.Position.Z) * hrp.CFrame.Rotation

			skySavedY = hrp.Position.Y
			skyAnchorX = hrp.Position.X
			skyAnchorZ = hrp.Position.Z
			skyAnchorRot = hrp.CFrame.Rotation

					return

				skyAnchorZ = curHrp.Position.Z
				skyAnchorRot = curHrp.CFrame.Rotation
				EnsureSkyFarmPlatform(curHrp.Position.X, curH, curHrp.Position.Z)
				if math.abs(curHrp.Position.Y - curH) > 15 then
					pcall(function()
						curHrp.AssemblyLinearVelocity = Vector3.zero
					end)
					curHrp.CFrame = CFrame.new(curHrp.Position.X, curH, curHrp.Position.Z) * curHrp.CFrame.Rotation
				end
			end)
		end
	end

		if Config.SkyFarm and not (IsCheckingSacredBeast or IsBuyingSecretRod or Config.AutoBoss or Config.AutoSecretBoss) then

		end
	end

	DestroySkyPlatform()
end

local function GetMaterialFishCount(fishName, minKg, maxKg)
	local pData = PlayerData or (DataFolder and DataFolder:FindFirstChild(tostring(LocalPlayer.UserId)))
	if not pData then
		return 0
	end
	local inv = pData:FindFirstChild("Inventory")
	if not inv then
		return 0
	end
	local count = 0
	local gsMod = nil
	pcall(function()
		local cm = ReplicatedStorage:FindFirstChild("ClientModule")
		local gs = cm and cm:FindFirstChild("GameService")
		if gs then
			gsMod = require(gs)
		end
	end)
	for _, entry in ipairs(inv:GetChildren()) do
		local rawName = entry.Name
		local cleanName = string.split(rawName, "| Favorite")[1]
		cleanName = string.gsub(cleanName, "^%s*(.-)%s*$", "%1")
		local matchedName = (rawName == fishName) or (cleanName == fishName)
		local weightNum = nil
		if gsMod and gsMod.DestringFish then
			local ok, res = pcall(gsMod.DestringFish, cleanName)
			if ok and res then
				if res.FishName and res.FishName == fishName then
					matchedName = true
				end
				if res.Weight then
					weightNum = tonumber(res.Weight)
				end
			end
		end
		if not weightNum then
			if entry:IsA("StringValue") then
				local parts = {}
				for p in string.gmatch(entry.Value, "[^|]+") do
					table.insert(parts, p)
				end
				if #parts >= 2 then
					weightNum = tonumber(parts[2]:gsub("^%s+", ""):gsub("%s+$", ""))
				elseif #parts == 1 then
					weightNum = tonumber(parts[1]:gsub("^%s+", ""):gsub("%s+$", ""))
				end
			elseif entry:IsA("NumberValue") or entry:IsA("IntValue") then
				weightNum = tonumber(entry.Value)
			end
		end
		if not weightNum then
			for num in string.gmatch(cleanName, "(%d+%.?%d*)%s*[kK][gG]") do
				weightNum = tonumber(num)
				break
			end
		end
		if matchedName and weightNum and weightNum >= minKg and weightNum <= maxKg then
			count = count + 1
		end
	end
	return count
end

		if string.find(string.lower(fishName), string.lower(k), 1, true) or string.find(string.lower(k), string.lower(fishName), 1, true) then
			return v
		end
	end
	return "Beginning Isle"
end

local isMaterialFarmCastingAllowed = false

local function IsWeatherMatching(curWeather, weatherTable)
	if not weatherTable then
		return true
	end
	local lowerW = string.lower(tostring(curWeather or ""))
	if lowerW == "" then
		return false
	end
	for k, _ in pairs(weatherTable) do
		local lowerK = string.lower(tostring(k))
		if lowerW == lowerK or string.find(lowerW, lowerK, 1, true) or string.find(lowerK, lowerW, 1, true) then
			return true
		end
	end
	return false
end

	if not Config.AutoFarmMaterialBait or Config.SkyFarm or IsCheckingSacredBeast or IsBuyingSecretRod then
		isMaterialFarmCastingAllowed = false
		currentMaterialTargets = {}

		isMaterialFarmCastingAllowed = false

		else
			isMaterialFarmCastingAllowed = true
		end

		return
	end

		local targetCF = IslandFarmPositions["Mystic Reef Isle"] or GetIslandFarmCFrame("Mystic Reef Isle")
		if targetCF and (hrp.Position - targetCF.Position).Magnitude > 5 then
			isMaterialFarmCastingAllowed = false
			if c:GetAttribute("Fishing") then
				pcall(function()
					FishingEvent:FireServer(hrp.CFrame)
				end)
				task.wait(0.1)
			end
			hrp.CFrame = targetCF
		else
			isMaterialFarmCastingAllowed = true
		end
		currentMaterialTargets = {}
		return
	end
	local recipe = BAIT_RECIPES[selectedBait]
	if not recipe or #recipe == 0 then
		isMaterialFarmCastingAllowed = false
		currentMaterialTargets = {}
		return
	end

			table.insert(availableList, item)
		end

		currentMaterialTargets = {}
		if c:GetAttribute("Fishing") then
			pcall(function()
				FishingEvent:FireServer(hrp.CFrame)
			end)

		if now - lastMatNotifyTick >= 6 then
			lastMatNotifyTick = now
			pcall(function()
				Fluent:Notify({
					Title = "Material Bait",
					Content = selectedBait .. " waiting for weather...",
					Duration = 3,
				})
			end)
		end
		return
	end

	local targetFish = nil
	local minCount = math.huge
	for _, item in ipairs(availableList) do
		local cnt = GetMaterialFishCount(item.name, item.minKg, item.maxKg)
		if cnt < minCount then
			minCount = cnt
			targetFish = item
		end
	end
	if not targetFish then
		isMaterialFarmCastingAllowed = false
		currentMaterialTargets = {}
		return
	end

	local islandName
	local reqWeather = MAT_WEATHER_REQUIRED[targetFish.name]
	if reqWeather then
		islandName = reqWeather.island
	else
		islandName = GetIslandForMaterialFish(targetFish.name)
	end

	local matTargets = {}
	for _, item in ipairs(availableList) do
		local isl = GetIslandForMaterialFish(item.name)
		if isl == islandName then
			table.insert(matTargets, string.lower(item.name))
			if item.name == "Primordial Kunfish Overlord" then
				table.insert(matTargets, "primordial kunfish")
			end
		end
	end
	if #matTargets == 0 and targetFish then
		table.insert(matTargets, string.lower(targetFish.name))
		if targetFish.name == "Primordial Kunfish Overlord" then
			table.insert(matTargets, "primordial kunfish")
		end
	end
	currentMaterialTargets = matTargets

	local targetCF = IslandFarmPositions[islandName] or GetIslandFarmCFrame(islandName)
	if targetCF then
		if (hrp.Position - targetCF.Position).Magnitude > 5 then
			isMaterialFarmCastingAllowed = false
			if c:GetAttribute("Fishing") then
				pcall(function()
					FishingEvent:FireServer(hrp.CFrame)
				end)
				task.wait(0.1)
			end
			hrp.CFrame = targetCF
		else
			isMaterialFarmCastingAllowed = true
		end
	else
		isMaterialFarmCastingAllowed = false
	end
end

local function FindSkillNumberValue(char, key)
	local skillsFolder = cachedSkillsFolder or (char and char:FindFirstChild("Skills"))

		return nil, nil
	end
	for _, s in ipairs(skillsFolder:GetChildren()) do
		if s:IsA("NumberValue") then
			local k = s:GetAttribute("Key") or s.Name
			if tostring(k) == key then
				return s, skillsFolder
			end
		end
	end
	return nil, skillsFolder
end

local function FindSkillButton(key)
	if not SkillBtnFrame then
		return nil
	end
	local direct = SkillBtnFrame:FindFirstChild(key)
	if direct then
		return direct
	end
	for _, child in ipairs(SkillBtnFrame:GetChildren()) do

				return child
			end
			for _, desc in ipairs(child:GetDescendants()) do
				if desc:IsA("TextLabel") and desc.Name ~= "CD" and desc.Text == key then
					return child
				end
			end
		end

	return nil
end

local function IsGameAllowingSkill(char)
	if not FishingGui or not FishingGui.Visible then
		wasMinigameVisible = false
		return false
	end

	if not wasMinigameVisible then
		wasMinigameVisible = true
		minigameOpenTick = os.clock()
	end

	if (RhythmGui and RhythmGui.Visible) or (BossFightBar and BossFightBar.Visible) then
		return false
	end

	if char:GetAttribute("Phase2") == true then
		bossPhase2SetupUntil = 0
		return false
	end

	if bossPhase2SetupUntil > 0 and os.clock() < bossPhase2SetupUntil then
		return false
	end

	if char:GetAttribute("SkillLocked") or char:GetAttribute("Confused") then
		return false
	end

	local stats = cachedStatsFolder or char:FindFirstChild("Stats")
	if stats and stats:FindFirstChild("Stun") and stats.Stun.Value == true then
		return false
	end

	local fishId = LocalPlayer:GetAttribute("FishID")
	local fishObj = (fishId and FishesFolder) and FishesFolder:FindFirstChild(tostring(fishId)) or nil
	if fishObj then
		local readyVal = fishObj:FindFirstChild("Ready")
		if readyVal then
			if readyVal.Value ~= true then
				return false
			end
		elseif os.clock() - minigameOpenTick < 3.1 then
			return false
		end
		if fishObj:GetAttribute("FinalPhase") == true or fishObj:GetAttribute("PreFinalPhase") == true then
			return false
		end
	elseif os.clock() - minigameOpenTick < 3.1 then
		return false
	end

	local skillsFolder = cachedSkillsFolder or char:FindFirstChild("Skills")
	if skillsFolder then
		for _, s in ipairs(skillsFolder:GetChildren()) do
			if s:IsA("NumberValue") and (s:GetAttribute("UsingSkill") or 0) > 0 then
				return false
			end
		end
	end

	return true
end

local function IsSkillSlotAvailable(char, key)
	if PlayerData then
		local rodVal = PlayerData:FindFirstChild("FishingRod")
		local rodInv = PlayerData:FindFirstChild("FishingRodInventory")
		local rf = (rodVal and rodVal.Value ~= "" and rodInv) and rodInv:FindFirstChild(rodVal.Value) or nil
		local skFolder = rf and rf:FindFirstChild("Skill")
		local skVal = skFolder and skFolder:FindFirstChild(key)
		if skVal and skVal:IsA("StringValue") then
			return skVal.Value ~= ""
		end
	end
	local btn = FindSkillButton(key)
	if btn then
		return btn.Visible == true
	end
	local s = FindSkillNumberValue(char, key)
	return s ~= nil
end

local function IsSkillOnCooldown(char, key)
	local btn = FindSkillButton(key)
	if btn then
		local cd = btn:FindFirstChild("CD")
		if cd and cd.Visible then
			local txt = cd.Text or ""
			local num = tonumber(string.match(txt, "%d+%.?%d*"))
			if num and num > 0 then
				return true
			end
		end
	end
	local s = FindSkillNumberValue(char, key)
	if s and (s.Value > 0 or (s:GetAttribute("UsingSkill") or 0) > 0) then
		return true
	end
	return false
end

local function IsSkillConfirmedUsed(char, key)
	return IsSkillOnCooldown(char, key)
end

local function IsSkillKeyReady(char, key)
	if not IsSkillSlotAvailable(char, key) then
		return false
	end

	if IsSkillOnCooldown(char, key) then
		return false
	end

	local btn = FindSkillButton(key)
	if btn then
		if not btn.Visible then
			return false
		end
		local lock = btn:FindFirstChild("Lock")
		if lock and lock.Visible then
			return false
		end
	end

	local s, skillsFolder = FindSkillNumberValue(char, key)
	if s then
		return true
	end

	if skillsFolder and #skillsFolder:GetChildren() > 0 then
		return false
	end

	return btn ~= nil and btn.Visible == true
end

			or Config.SkyFarm

		FishingBar.Position = CENTER_BAR_POS

		barPosConn = FishingBar:GetPropertyChangedSignal("Position"):Connect(function()
			if not isLockingBar and FishingBar.Position.X.Scale ~= 0.5 then
				LockFishingBar()
			end
		end)
	end

local function TriggerGuiButton(btn)
	if not btn then
		return false
	end
	if not btn:IsA("GuiButton") then
		btn = btn:FindFirstChildWhichIsA("GuiButton", true)
			or (btn.Parent and btn.Parent:IsA("GuiButton") and btn.Parent)
			or nil
		if not btn then
			return false
		end
	end
	local triggered = false
	local sigNames = { "MouseButton1Click", "Activated", "MouseButton1Down", "MouseButton1Up" }
	if getconnections then
		for _, sName in ipairs(sigNames) do
			local okSig, sig = pcall(function()
				return btn[sName]
			end)
			if okSig and sig then
				pcall(function()
					for _, conn in pairs(getconnections(sig)) do
						if conn.Function then
							task.spawn(conn.Function)
							triggered = true
						elseif conn.Fire then
							conn:Fire()
							triggered = true
						end
					end
				end)
			end
		end
	end
	if firesignal then
		for _, sName in ipairs(sigNames) do
			pcall(function()
				firesignal(btn[sName])
				triggered = true
			end)
		end
	end
	return triggered
end

local getupval = (debug and debug.getupvalue) or getupvalue

local function StepPhase2MiniGame()
	if not BossFightBar or not BossFightBar.Visible then
		if #phase2BarHistory > 0 then
			table.clear(phase2BarHistory)
		end
		return
	end

	local bar = BossFightBar:FindFirstChild("Bar")
	local hitbox = BossFightBar:FindFirstChild("Hitbox")
	if not bar or not hitbox then
		return
	end

	local barW = bar.Size.X.Scale
	local barCenter = bar.Position.X.Scale + barW * (0.5 - bar.AnchorPoint.X)
	local hbW = hitbox.Size.X.Scale
	local hbLeft = hitbox.Position.X.Scale - hbW * hitbox.AnchorPoint.X

	local now = os.clock()
	table.insert(phase2BarHistory, { t = now, x = barCenter })
	while #phase2BarHistory > 1 and (now - phase2BarHistory[1].t) > 0.15 do
		table.remove(phase2BarHistory, 1)
	end

	local hitDelay = (tonumber(Config.BossHitDelay) or 15) / 1000
	if hitDelay < 0.005 then
		hitDelay = 0.005
	end

	local handledViaUpval = false
	local onClickFn = nil

	if getconnections and getupval and MobileFishingBtn then
		pcall(function()
			for _, conn in pairs(getconnections(MobileFishingBtn.MouseButton1Down)) do
				local fn = conn.Function and getupval(conn.Function, 1)
				if type(fn) == "function" then
					onClickFn = fn
					local u12 = getupval(fn, 1)
					local u14 = getupval(fn, 2)
					local u27 = getupval(fn, 3)
					local u26 = getupval(fn, 4)
					local u1 = getupval(fn, 5)
					local u25 = getupval(fn, 6)
					local u23 = getupval(fn, 7)
					local u2 = getupval(fn, 8)
					local u15 = getupval(fn, 9)
					if
						u12 == true
						and type(u14) == "number"
						and type(u27) == "number"
						and type(u26) == "number"
						and type(u1) == "number"
						and type(u25) == "number"
						and type(u23) == "number"
						and type(u2) == "number"
						and type(u15) == "number"
					then
						if math.abs(u2 - hbW) < 0.005 and math.abs(u15 - hbLeft) < 0.005 then
							handledViaUpval = true
							if (now - u14 >= hitDelay or u14 == 0) and now - lastPhase2HitTick >= hitDelay then
								local v91 = (now - 0.025 - u27) % u26 * u1
								if u25 < v91 then
									v91 = u25 * 2 - v91
								end
								local v92 = u23 + v91
								local v93 = u2 * 0.3
								if v92 >= (u15 - v93 * 0.8) and v92 <= (u15 + u2 + v93 * 0.8) then
									lastPhase2HitTick = now
									fn(false)
								end
							end
						end
						break
					end
				end
			end
		end)
	end

	if handledViaUpval then
		return
	end

	if now - lastPhase2HitTick < hitDelay then
		return
	end

	local evalX = barCenter
	local pad = hbW * 0.2

	local badBox = BossFightBar:FindFirstChild("HitboxBad")
	if badBox and badBox.Visible then
		local badW = badBox.Size.X.Scale
		local badLeft = badBox.Position.X.Scale - badW * badBox.AnchorPoint.X
		if evalX >= (badLeft - 0.01) and evalX <= (badLeft + badW + 0.01) then
			return
		end
	end

	if evalX >= (hbLeft - pad) and evalX <= (hbLeft + hbW + pad) then
		lastPhase2HitTick = now
		local clicked = false
		if onClickFn then
			pcall(function()
				onClickFn(false)
				clicked = true
			end)
		end
		if not clicked and MobileFishingBtn then
			if getconnections then
				pcall(function()
					for _, conn in pairs(getconnections(MobileFishingBtn.MouseButton1Down)) do
						if conn.Function then
							task.spawn(conn.Function)
							clicked = true
						elseif conn.Fire then
							conn:Fire()
							clicked = true
						end
					end
				end)
			end
			if not clicked and firesignal then
				pcall(function()
					firesignal(MobileFishingBtn.MouseButton1Down)
					clicked = true
				end)
			end
		end
		if not clicked then
			pcall(function()
				VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 0)
				task.delay(0.005, function()
					VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0)
				end)
			end)
		end
	end
end

local function IsTrapNote(gui)
	if not gui or not gui:IsA("GuiObject") then
		return false
	end
	if gui:GetAttribute("Trap") == true or gui:GetAttribute("trap") == true or gui:GetAttribute("IsTrap") == true then
		return true
	end
	if string.find(string.lower(gui.Name), "trap", 1, true) then
		return true
	end
	local function isRed(c)
		return c and c.R > 0.65 and c.G < 0.45 and c.B < 0.45 and c.R > c.G * 1.8 and c.R > c.B * 1.8
	end
	if isRed(gui.BackgroundColor3) then
		return true
	end
	if (gui:IsA("ImageLabel") or gui:IsA("ImageButton")) and isRed(gui.ImageColor3) then
		return true
	end
	for _, desc in ipairs(gui:GetDescendants()) do
		if desc:IsA("GuiObject") then
			if isRed(desc.BackgroundColor3) then
				return true
			end
			if (desc:IsA("ImageLabel") or desc:IsA("ImageButton")) and isRed(desc.ImageColor3) then
				return true
			end
		end
	end
	return false
end

local function StepRhythmMiniGame()
	if not RhythmGui or not RhythmGui.Visible then
		return
	end
	local now = os.clock()
	if now - lastRhythmScan < 0.025 then
		return
	end
	lastRhythmScan = now

	for _, lane in ipairs(rhythmLanes) do
		if now - (lastRhythmLaneHit[lane] or 0) >= 0.05 then
			local laneFrame = RhythmGui:FindFirstChild("Progression" .. lane)
			if laneFrame then
				local barFrame = laneFrame:FindFirstChild("BarFrame")
				local targetY = barFrame and barFrame.Position.Y.Scale or 0.85
				local closestNote = nil
				local closestDist = 999
				for _, child in ipairs(laneFrame:GetChildren()) do
					if child.Name == "Note_FX" and child:IsA("GuiObject") and child.Visible then
						local dist = math.abs(child.Position.Y.Scale - targetY)
						if dist < closestDist then
							closestDist = dist
							closestNote = child
						end
					end
				end
				if closestNote and not IsTrapNote(closestNote) and closestDist <= 0.18 then
					lastRhythmLaneHit[lane] = now
					local btn = laneFrame:FindFirstChild("Button")
					if btn then
						TriggerGuiButton(btn)
					end
					if closestNote.Parent ~= nil then
						local kc = rhythmKeyMap[lane]
						if kc then
							pcall(function()
								VirtualInputManager:SendKeyEvent(true, kc, false, game)
								task.delay(0.02, function()
									VirtualInputManager:SendKeyEvent(false, kc, false, game)
								end)
							end)
						end
					end
					if closestNote.Parent ~= nil and closestDist <= 0.1 and RhythmHitEvent then
						pcall(function()
							closestNote:Destroy()
							RhythmHitEvent:FireServer("hit")
						end)
					end
				end
			end
		end
	end
end

		return true
	end
	local rawName = tostring(obj:GetAttribute("Name") or obj:GetAttribute("FishName") or obj.Name)
	local lowerName = string.lower(rawName)
	if
		string.find(lowerName, "secret")
		or string.find(lowerName, "long diaohai")
		or string.find(lowerName, "longdiaohai")
		or string.find(lowerName, "diaohai")
		or string.find(lowerName, "wuzhi")
	then
		return true
	end
	local inv = Info:FindFirstChild("Inventory")
	local mod = inv and inv:FindFirstChild(rawName)
	local data = GetModuleData(mod)
	if
		data
		and (
			data.Secret == true
			or data.Rarity == "Secret"
			or (data.Boss == true and (tonumber(data.Chance) or 100) <= 1)
		)
	then
		return true
	end
	return false
end

local function FindBossTarget(wantSecret)
	local npcFolder = workspace:FindFirstChild("NPC")
	local bossFolder = npcFolder and npcFolder:FindFirstChild("Boss")

	if bossFolder then
		for _, boss in ipairs(bossFolder:GetChildren()) do
			local part = GetModelPart(boss)
			if part then
				local isSec = IsSecretBossEntity(boss)
				if wantSecret and isSec then
					return boss, part
				elseif not wantSecret and not isSec then
					return boss, part
				end
			end
		end
	end

	if not wantSecret then
		local wsBoss = workspace:FindFirstChild("Boss") or workspace:FindFirstChild("BossZone")
		if wsBoss then
			local part = GetModelPart(wsBoss)
			if part then
				return wsBoss, part
			end
		end
	end

	return nil, nil
end

local function ParseBossCooldownSeconds(text)
	if not text or text == "" then
		return nil
	end
	local lower = string.lower(text)
	local m, s = string.match(lower, "(%d+)%s*:%s*(%d+)")
	if m and s then
		return (tonumber(m) or 0) * 60 + (tonumber(s) or 0)
	end
	local mPart = string.match(lower, "(%d+)%s*m")
	local sPart = string.match(lower, "(%d+)%s*s")
	if mPart or sPart then
		return (tonumber(mPart) or 0) * 60 + (tonumber(sPart) or 0)
	end
	local secOnly = string.match(lower, "(%d+)%s*sec")
	if secOnly then
		return tonumber(secOnly) or 0
	end
	return nil
end

local function CheckBossCooldownText(text)
	if not text or text == "" then
		return
	end
	local lower = string.lower(text)
	local isEnzo = string.find(lower, "enzo", 1, true) ~= nil
	local isLong = (string.find(lower, "long diaohai", 1, true) ~= nil)
		or (string.find(lower, "longdiaohai", 1, true) ~= nil)
		or (string.find(lower, "diaohai", 1, true) ~= nil)

	if not isEnzo and not isLong then
		return
	end

	local now = os.clock()
	if string.find(lower, "cooldown", 1, true) then
		local sec = ParseBossCooldownSeconds(lower)
		if sec and sec > 0 and sec <= 3600 then
			if isEnzo then
				enzoCooldownUntil = now + sec
			elseif isLong then
				longDiaohaiCooldownUntil = now + sec
			end
		end
	elseif string.find(lower, "already being fought", 1, true) then
		if isEnzo then
			if now >= enzoCooldownUntil then
				enzoCooldownUntil = now + 5
			end
		elseif isLong then
			if now >= longDiaohaiCooldownUntil then
				longDiaohaiCooldownUntil = now + 5
			end
		end
	end
end

local function StepAutoBoss(char, isMinigame)
	if isMinigame then
		return
	end

	local enzoOn = Config.AutoBoss == true
	local longOn = Config.AutoSecretBoss == true

	if not enzoOn and not longOn then
		return
	end

	local now = os.clock()
	if now - lastBossActionTick < 0.5 then
		return
	end
	lastBossActionTick = now

	local enzoReady = enzoOn and (now >= enzoCooldownUntil)
	local longReady = longOn and (now >= longDiaohaiCooldownUntil)

	local targetBoss = nil
	if enzoOn and longOn then
		if enzoReady and not longReady then
			targetBoss = "Enzo"
		elseif longReady and not enzoReady then
			targetBoss = "Long Diaohai"
		elseif enzoReady and longReady then
			if currentBossTarget == "Long Diaohai" then
				targetBoss = "Long Diaohai"
			else
				targetBoss = "Enzo"
			end
		else
			if enzoCooldownUntil <= longDiaohaiCooldownUntil then
				targetBoss = "Enzo"
			else
				targetBoss = "Long Diaohai"
			end
		end
	elseif enzoOn then
		targetBoss = "Enzo"
	elseif longOn then
		targetBoss = "Long Diaohai"
	end

	if not targetBoss then
		return
	end

	currentBossTarget = targetBoss

	local isTargetReady = (targetBoss == "Long Diaohai") and (now >= longDiaohaiCooldownUntil) or (now >= enzoCooldownUntil)

	local c = char or cachedChar or LocalPlayer.Character
	if c and c:GetAttribute("Type") ~= "Fishing Rod" then
		pcall(function()
			ToggleHotbar:InvokeServer("1")
		end)
	end

	if isTargetReady then
		pcall(function()
			local args = {
				[1] = targetBoss,
				[2] = Config.BossMode or "Normal",
			}
			game:GetService("ReplicatedStorage"):WaitForChild("Events", 9e9):WaitForChild("StartBossFight", 9e9):FireServer(unpack(args))
		end)
	end
end

				pcall(function()
					FishingGui.Visible = false
				end)
				isMinigame = false
				ghostMinigameCheckTick = 0
			end
		else
			ghostMinigameCheckTick = 0
		end
		if wasMinigameActive and not isMinigame then
			lastMinigameEndTick = now
		end
		wasMinigameActive = isMinigame
		local isMaterialFarmActive = Config.AutoFarmMaterialBait and isMaterialFarmCastingAllowed
		local activeReel = Config.AutoReel
			or Config.AutoMiniGame
			or Config.AutoBoss
			or Config.AutoSecretBoss
			or Config.AutoFarmIsland
			or Config.SkyFarm
			or isMaterialFarmActive

			or Config.AutoFarmIsland
			or Config.SkyFarm
			or isMaterialFarmActive

			activeCast = false
		end

			or (Config.AutoGiveEventFish and HasPendingEventFish and HasPendingEventFish())
		if needPauseForGive and not isMinigame then
			activeCast = false
		end
		local activeSkill = Config.AutoSkill
			or Config.AutoBoss
			or Config.AutoSecretBoss
			or Config.AutoFarmIsland
			or Config.SkyFarm
			or isMaterialFarmActive

		if not isMinigame then
			skillSeqIndex = 1
			pendingSkillKey = nil
		end

		if activeReel and isMinigame then
			pcall(function()
				UpdateFishProgression:FireServer()
			end)
		end

		if activeSkill and isMinigame then
			local char = cachedChar
			local order = Config.SkillOrder
			local orderCount = type(order) == "table" and #order or 0
			if char and orderCount > 0 then
				if skillSeqIndex < 1 or skillSeqIndex > orderCount then
					skillSeqIndex = 1
				end

				local readyIdx = nil
				for step = 1, orderCount do
					local idx = ((skillSeqIndex - 1 + (step - 1)) % orderCount) + 1
					local k = order[idx]
					if IsSkillKeyReady(char, k) then
						readyIdx = idx
						break
					end
				end

				if readyIdx then
					skillSeqIndex = readyIdx
					local currentKey = order[skillSeqIndex]
					if IsGameAllowingSkill(char) and (now - lastSkillTry >= 0.25) then
						lastSkillTry = os.clock()
						pendingSkillKey = currentKey
						pcall(function()
							UseSkill:FireServer(currentKey)
						end)
						skillSeqIndex = (skillSeqIndex % orderCount) + 1
					end
				else
					for step = 1, orderCount do
						local idx = ((skillSeqIndex - 1 + (step - 1)) % orderCount) + 1
						if IsSkillSlotAvailable(char, order[idx]) and not IsSkillOnCooldown(char, order[idx]) then
							skillSeqIndex = idx
							break
						end
					end
				end
			end
		end

		if Config.AutoBoss or Config.AutoSecretBoss then
			StepAutoBoss(cachedChar, isMinigame)
		else
			if not isMinigame then
				StepAutoTPPos(cachedChar)
				StepAutoFarmIsland(cachedChar)
				StepAutoFarmMaterialBait(cachedChar)
			end
			if activeCast and not isMinigame then
				local minCastInterval = (Config.AutoCastMethod == "Instant") and 0.05
					or math.clamp(tonumber(Config.CastDelay) or 1, 0, 10)
				if (now - lastMinigameEndTick >= minCastInterval) and (now - lastCastTry >= minCastInterval) then
					local char = cachedChar
					if char then
						if char:GetAttribute("Type") ~= "Fishing Rod" then
							lastCastTry = now
							pcall(function()
								ToggleHotbar:InvokeServer("1")
							end)
						elseif not char:GetAttribute("Fishing") then
							local hrp = cachedHrp or char:FindFirstChild("HumanoidRootPart")
							if hrp then
								lastCastTry = now
								pcall(function()
									FishingEvent:FireServer(hrp.CFrame)
								end)
							end
						end
					end
				end
			end
		end
	end
end)

local function FindAllGodSpirits()
	local results = {}
	local seen = {}

	local function resolveEntityTarget(desc)
		local part = GetModelPart(desc)
		if part and part:IsDescendantOf(workspace) then
			return part.Position, part.CFrame, part, desc
		end
		if desc:IsA("PVInstance") then
			local ok, cf = pcall(function()
				return desc:GetPivot()
			end)
			if ok and cf and cf.Position.Magnitude > 1 then
				return cf.Position, cf, nil, desc
			end
		end
		return nil, nil, nil, nil
	end

	local function checkObj(desc)
		if not desc or desc:IsA("Folder") then
			return
		end
		if not (desc:IsA("Model") or desc:IsA("BasePart")) then
			return
		end
		local dName = tostring(desc.Name or "")
		local aName = tostring(desc:GetAttribute("Name") or "")
		if dName == "God" or dName == "Lamp" or aName == "God" or aName == "Lamp" then
			return
		end
		local dLower = string.lower(string.gsub(dName, "%s+", ""))
		local aLower = string.lower(string.gsub(aName, "%s+", ""))

		local col = nil
		if string.find(dLower, "yellow", 1, true) or string.find(aLower, "yellow", 1, true) then
			col = "yellow"
		elseif string.find(dLower, "green", 1, true) or string.find(aLower, "green", 1, true) then
			col = "green"
		elseif string.find(dLower, "blue", 1, true) or string.find(aLower, "blue", 1, true) then
			col = "blue"
		end

		local isInsideGod = false
		local p = desc.Parent
		while p and p ~= workspace do
			if p.Name == "God" then
				isInsideGod = true
				break
			end
			p = p.Parent
		end

		if not col and isInsideGod then
			col = dLower ~= "" and dLower or "yellow"
		end

		if col and not seen[col] then
			local pos, cf, part, obj = resolveEntityTarget(desc)
			if pos then
				seen[col] = true
				table.insert(results, {
					name = (dName ~= "" and dName ~= "God") and dName or col,
					part = part,
					obj = obj,
					pos = pos,
					cf = cf,
					lower = col,
				})
			end
		end
	end

	local npcGod = workspace:FindFirstChild("NPC") and workspace.NPC:FindFirstChild("God")
	if npcGod then
		for _, child in ipairs(npcGod:GetChildren()) do
			checkObj(child)
		end
		if #results > 0 then
			return results
		end
	end

	local godFolder = workspace:FindFirstChild("God")
	if godFolder then

local function FindSpecialEntities()
	local candidatesMaoshan = {}
	local candidatesTaoist = {}

	local function checkCandidate(desc, folderName)
		if not desc or desc:IsA("Folder") then
			return
		end
		if not (desc:IsA("Model") or desc:IsA("BasePart")) then
			return
		end
		local dName = tostring(desc.Name or "")
		local aName = tostring(desc:GetAttribute("Name") or "")
		if dName == "God" or dName == "Lamp" or (dName == "" and aName == "") then
			return
		end
		local part = GetModelPart(desc)
		local pos = nil
		local cf = nil
		if part and part:IsDescendantOf(workspace) then
			pos = part.Position
			cf = part.CFrame
		elseif desc:IsA("PVInstance") then
			local ok, pcf = pcall(function()
				return desc:GetPivot()
			end)
			if ok and pcf and pcf.Position.Magnitude > 1 then
				pos = pcf.Position
				cf = pcf
			end
		end
		if not pos then
			return
		end

		local dLower = string.lower(string.gsub(dName, "%s+", ""))
		local aLower = string.lower(string.gsub(aName, "%s+", ""))

		local isMao = string.find(dLower, "maoshan", 1, true)
			or string.find(dLower, "maosan", 1, true)
			or string.find(aLower, "maoshan", 1, true)
			or string.find(aLower, "maosan", 1, true)

		local isTao = string.find(dLower, "taoist", 1, true)
			or string.find(dLower, "taosi", 1, true)
			or string.find(aLower, "taoist", 1, true)
			or string.find(aLower, "taosi", 1, true)

		if not isMao and not isTao then
			for _, attrVal in pairs(desc:GetAttributes()) do
				local valLower = string.lower(string.gsub(tostring(attrVal), "%s+", ""))
				if string.find(valLower, "maoshan", 1, true) or string.find(valLower, "maosan", 1, true) then
					isMao = true
					break
				elseif string.find(valLower, "taoist", 1, true) or string.find(valLower, "taosi", 1, true) then
					isTao = true
					break
				end
			end
		end

		if isMao then
			table.insert(candidatesMaoshan, { part = part, pos = pos, cf = cf, folder = folderName, obj = desc })
		elseif isTao then
			table.insert(candidatesTaoist, { part = part, pos = pos, cf = cf, folder = folderName, obj = desc })
		end
	end

	local wsNPC = workspace:FindFirstChild("NPC")
	if wsNPC then
		for _, desc in ipairs(wsNPC:GetDescendants()) do
			checkCandidate(desc, "NPC")
		end
	end

	local finalMaoshan = candidatesMaoshan[1] or nil
	local finalTaoist = candidatesTaoist[1] or nil

	local godSpirits = FindAllGodSpirits()
	return finalMaoshan, finalTaoist, godSpirits
end

local ActiveEntityESPs = {}

local function destroyEntityESPEntry(entry)
	if not entry then
		return
	end
	if type(entry) == "userdata" then
		pcall(function()
			entry:Destroy()
		end)
		return
	end
	if entry.gui then
		pcall(function()
			entry.gui:Destroy()
		end)
		entry.gui = nil
	end
	if entry.hl then
		pcall(function()
			entry.hl:Destroy()
		end)
		entry.hl = nil
	end
	if entry.att then
		pcall(function()
			entry.att:Destroy()
		end)
		entry.att = nil
	end
end

local function applyEntityESP(targetInfo, tag, title)
	if not targetInfo or not targetInfo.pos then
		return
	end
	local entityFill = Color3.fromRGB(170, 50, 255)
	local entityOutline = Color3.fromRGB(225, 165, 255)
	local entityText = Color3.fromRGB(195, 100, 255)
	if tag == "Taoist" then
		entityFill = Color3.fromRGB(255, 215, 0)
		entityOutline = Color3.fromRGB(255, 240, 150)
		entityText = Color3.fromRGB(255, 225, 50)
	elseif tag == "Maoshan" then
		entityFill = Color3.fromRGB(115, 118, 200)
		entityOutline = Color3.fromRGB(165, 170, 255)
		entityText = Color3.fromRGB(140, 145, 240)
	elseif string.find(tag, "blue") then
		entityFill = Color3.fromRGB(0, 150, 255)
		entityOutline = Color3.fromRGB(120, 210, 255)
		entityText = Color3.fromRGB(60, 180, 255)
	elseif string.find(tag, "yellow") then
		entityFill = Color3.fromRGB(255, 230, 0)
		entityOutline = Color3.fromRGB(255, 255, 150)
		entityText = Color3.fromRGB(255, 240, 50)
	elseif string.find(tag, "green") then
		entityFill = Color3.fromRGB(0, 230, 80)
		entityOutline = Color3.fromRGB(140, 255, 170)
		entityText = Color3.fromRGB(60, 255, 110)
	end
	local part = targetInfo.part
	local obj = targetInfo.obj or part
	local pos = targetInfo.pos

	local entry = ActiveEntityESPs[tag]
	if not entry or type(entry) ~= "table" then
		if entry then
			destroyEntityESPEntry(entry)
		end
		entry = {}
		ActiveEntityESPs[tag] = entry
	end

	local adornee = nil
	if part and part:IsDescendantOf(workspace) then
		adornee = part
		if entry.att then
			pcall(function()
				entry.att:Destroy()
			end)
			entry.att = nil
		end
	else
		local terrain = workspace:FindFirstChildOfClass("Terrain")
		if terrain then
			if not (entry.att and entry.att.Parent == terrain) then
				local att = Instance.new("Attachment")
				att.Name = "StarCrowEntityAtt_" .. tag
				att.Parent = terrain
				entry.att = att
			end
			entry.att.WorldPosition = pos
			adornee = entry.att
		end
	end

	if entry.hl then
		pcall(function()
			entry.hl:Destroy()
		end)
		entry.hl = nil
	end

	if adornee then
		if not (entry.gui and entry.gui.Parent) then
			local esp = Instance.new("BillboardGui")
			esp.Name = "StarCrowEntityESP"
			esp.Size = UDim2.new(0, 220, 0, 26)
			esp.StudsOffset = Vector3.new(0, 3.5, 0)
			esp.AlwaysOnTop = true
			esp.MaxDistance = math.huge
			local text = Instance.new("TextLabel")
			text.Name = "NameText"
			text.Parent = esp
			text.Size = UDim2.new(1, 0, 1, 0)
			text.BackgroundTransparency = 1
			text.TextScaled = false
			text.TextSize = 13
			text.Font = Enum.Font.GothamBold
			text.TextColor3 = entityText
			text.TextStrokeTransparency = 0
			text.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
			text.TextXAlignment = Enum.TextXAlignment.Center
			entry.gui = esp
		end
		entry.gui.Adornee = adornee
		if entry.gui.Parent ~= adornee then
			entry.gui.Parent = adornee
		end
		local nameLabel = entry.gui:FindFirstChild("NameText")
		if nameLabel then
			nameLabel.TextColor3 = entityText
			nameLabel.Text = title
		end
	end
end

local function clearEntityESP(tag)
	if tag then
		if tag == "GodSpirit" then
			for k, entry in pairs(ActiveEntityESPs) do
				if string.find(k, "GodSpirit_") then
					destroyEntityESPEntry(entry)
					ActiveEntityESPs[k] = nil
				end
			end
		else
			if ActiveEntityESPs[tag] then
				destroyEntityESPEntry(ActiveEntityESPs[tag])
				ActiveEntityESPs[tag] = nil
			end
		end
	else
		for k, entry in pairs(ActiveEntityESPs) do
			destroyEntityESPEntry(entry)
			ActiveEntityESPs[k] = nil
		end
	end
end

local function updateEntityESP(finalMaoshan, finalTaoist, godSpirits)
	if Config.ESPMaoshan and finalMaoshan then
		applyEntityESP(finalMaoshan, "Maoshan", "Maoshan")
	else
		clearEntityESP("Maoshan")
	end

	if Config.ESPTaoist and finalTaoist then
		applyEntityESP(finalTaoist, "Taoist", "Taoist")
	else
		clearEntityESP("Taoist")
	end

	if Config.ESPGodSpirit and #godSpirits > 0 then
		local currentKeys = {}
		for _, g in ipairs(godSpirits) do
			local tag = "GodSpirit_" .. g.lower
			currentKeys[tag] = true
			local dispName = g.lower == "yellow" and "Yellow"
				or (g.lower == "green" and "Green" or (g.lower == "blue" and "Blue" or (string.upper(string.sub(g.lower, 1, 1)) .. string.sub(g.lower, 2))))
			applyEntityESP(g, tag, dispName)
		end
		for k, _ in pairs(ActiveEntityESPs) do
			if string.find(k, "GodSpirit_") and not currentKeys[k] then
				clearEntityESP(k)
			end
		end
	else
		for k, _ in pairs(ActiveEntityESPs) do
			if string.find(k, "GodSpirit_") then
				clearEntityESP(k)
			end
		end
	end
end
